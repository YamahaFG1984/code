第一个测试

📋 需求：maxBlock

给定一个字符串，返回其中最长的「块」的长度。块是由相同字符组成的连续序列。例如 "aabbbcc" 的最长块是 "bbb"，返回 3；空字符串返回 0。

public class BlockCounter {

  public int maxBlock(String str) {
    if (str.isEmpty())
      return 0;

    int largestBlockSize = 1;
    int currentBlockSize = 1;
    char lastChar = str.charAt(0);

    for (int i = 1; i < str.length(); i++) {
      char currentChar = str.charAt(i);
      // 和上一个字符相同，块变长；否则开始一个新块
      currentBlockSize = (currentChar == lastChar ? currentBlockSize + 1 : 1);
      lastChar = currentChar;
      largestBlockSize = Math.max(largestBlockSize, currentBlockSize);
    }
    return largestBlockSize;
  }
}

def max_block(s: str) -> int:
    if not s:
        return 0

    largest_block_size = 1
    current_block_size = 1
    last_char = s[0]

    for current_char in s[1:]:
        # 和上一个字符相同，块变长；否则开始一个新块
        current_block_size = current_block_size + 1 if current_char == last_char else 1
        last_char = current_char
        largest_block_size = max(largest_block_size, current_block_size)
    return largest_block_size

export function maxBlock(str: string): number {
  if (str.length === 0) return 0;

  let largestBlockSize = 1;
  let currentBlockSize = 1;
  let lastChar = str[0];

  for (let i = 1; i < str.length; i++) {
    const currentChar = str[i];
    // 和上一个字符相同，块变长；否则开始一个新块
    currentBlockSize = currentChar === lastChar ? currentBlockSize + 1 : 1;
    lastChar = currentChar;
    largestBlockSize = Math.max(largestBlockSize, currentBlockSize);
  }
  return largestBlockSize;
}

💡 JUnit 自带断言 vs AssertJ

JUnit 自己也有 assertEquals(expected, actual)，但参数顺序容易写反（写反了失败信息就会误导你）。 本书统一用 AssertJ 的 assertThat(actual).isEqualTo(expected)：读起来像句子，而且提供了大量专门的断言（集合、异常、字符串……）。 pytest 直接用 Python 的 assert，失败时会自动展示两边的值；Vitest 的 expect(actual).toBe(expected) 和 AssertJ 很像。

ⓘ 失败（failure）和错误（error）

JUnit 区分两者：失败是断言不成立（期望 2 实际 1）；错误是测试执行过程中抛出了意料之外的异常（比如越界）。两者都说明有问题，但错误往往意味着程序在更早的地方就崩了。

每个测试前后：准备与清理

上面每个测试都 new BlockCounter()，有点重复。JUnit 提供了生命周期注解，让你在测试前后执行代码：

public class BlockCounterWithBeforeAndAfterTest {

  private BlockCounter blockCounter;

  @BeforeEach                          // 每个 @Test 之前都会执行
  void setup() {
    this.blockCounter = new BlockCounter();
  }

  @Test
  void oneLongestBlock() {
    int blockSize = blockCounter.maxBlock("aabbbcc");
    assertThat(blockSize).isEqualTo(3);
  }

  @Test
  void emptyString() {
    int blockSize = blockCounter.maxBlock("");
    assertThat(blockSize).isEqualTo(0);
  }
  // ……其余测试同理
}

import pytest

class BlockCounter:                    # 假设有一个类版本，便于演示 fixture
    def max_block(self, s: str) -> int:
        return max_block(s)

@pytest.fixture                        # 每个用到它的测试都会拿到一个新实例
def block_counter():
    counter = BlockCounter()
    yield counter                      # yield 之前 = 准备；之后 = 清理（这里没什么可清理）

def test_one_longest_block(block_counter):   # 参数名和 fixture 同名，pytest 自动注入
    assert block_counter.max_block("aabbbcc") == 3

def test_empty_string(block_counter):
    assert block_counter.max_block("") == 0

import { beforeEach, test, expect } from "vitest";

class BlockCounter {                   // 假设有一个类版本，便于演示 beforeEach
  maxBlock(str: string) { return maxBlock(str); }
}

let blockCounter: BlockCounter;

beforeEach(() => {                     // 每个 test 之前都会执行
  blockCounter = new BlockCounter();
});

test("one longest block", () => {
  expect(blockCounter.maxBlock("aabbbcc")).toBe(3);
});

test("empty string", () => {
  expect(blockCounter.maxBlock("")).toBe(0);
});
💡 其实 JUnit 默认就会给每个测试一个新实例

JUnit 5 默认为每个测试方法创建一个新的测试类实例，所以直接写 private BlockCounter blockCounter = new BlockCounter(); 效果也一样（第 2、6 章的测试就是这么写的）。@BeforeEach 在准备工作比较复杂、或需要抛受检异常时更有用；@AfterEach 则用于关闭连接、删除文件等清理工作（第 9 章）。

参数化测试

六个测试的结构一模一样，只是输入和期望输出不同。参数化测试把它们写成一张表，测试逻辑只写一次：

public class BlockCounterParameterizedTest {

  @ParameterizedTest
  @CsvSource({                         // 每行一个用例，逗号分隔各个参数
      "aabbbcc,3",
      "aabbbccc,3",
      "abc,1",
      "aa,2",
      "aabbbb,4",
      "'',0"                           // 空字符串要写成 ''
  })
  void countTheNumberOfMaxBlocks(String input, int expectedOutput) {
    int blockSize = new BlockCounter().maxBlock(input);
    assertThat(blockSize).isEqualTo(expectedOutput);
  }
}

import pytest
from block_counter import max_block

@pytest.mark.parametrize("input_str, expected_output", [   # 每个元组一个用例
    ("aabbbcc", 3),
    ("aabbbccc", 3),
    ("abc", 1),
    ("aa", 2),
    ("aabbbb", 4),
    ("", 0),
])
def test_count_the_number_of_max_blocks(input_str, expected_output):
    assert max_block(input_str) == expected_output

import { test, expect } from "vitest";
import { maxBlock } from "./blockCounter";

test.each([                            // 每个数组一个用例
  ["aabbbcc", 3],
  ["aabbbccc", 3],
  ["abc", 1],
  ["aa", 2],
  ["aabbbb", 4],
  ["", 0],
])("maxBlock(%j) = %i", (input, expectedOutput) => {   // %j、%i 会被替换成参数，出现在报告里
  expect(maxBlock(input)).toBe(expectedOutput);
});

数据更复杂时（比如要传列表或对象），CSV 写不下了。JUnit 的 @MethodSource 让一个方法返回参数流 —— 第 2 章的 add() 就用了它：

public class BlockCounterParameterizedTest2 {

  @ParameterizedTest
  @MethodSource("inputs")              // 参数来自下面同名的静态方法
  void countTheNumberOfMaxBlocks(String input, int expectedOutput) {
    int blockSize = new BlockCounter().maxBlock(input);
    assertThat(blockSize).isEqualTo(expectedOutput);
  }

  static Stream<Arguments> inputs() {
    return Stream.of(
        Arguments.of("aabbbcc", 3),
        Arguments.of("aabbbccc", 3),
        Arguments.of("abc", 1),
        Arguments.of("aa", 2),
        Arguments.of("aabbbb", 4),
        Arguments.of("", 0));
  }
}

def inputs():                          # 普通函数返回用例列表，想怎么生成都行
    return [
        pytest.param("aabbbcc", 3, id="one-longest-block"),     # id：报告里显示的名字
        pytest.param("aabbbccc", 3, id="two-longest-blocks"),
        pytest.param("abc", 1, id="blocks-of-length-1"),
        pytest.param("aa", 2, id="single-block"),
        pytest.param("aabbbb", 4, id="longest-is-last"),
        pytest.param("", 0, id="empty"),
    ]

@pytest.mark.parametrize("input_str, expected_output", inputs())
def test_count_the_number_of_max_blocks(input_str, expected_output):
    assert max_block(input_str) == expected_output

const inputs = [                       // 用对象让每一列都有名字，比数组更清楚
  { name: "one longest block", input: "aabbbcc", expected: 3 },
  { name: "two longest blocks", input: "aabbbccc", expected: 3 },
  { name: "blocks of length 1", input: "abc", expected: 1 },
  { name: "single block", input: "aa", expected: 2 },
  { name: "longest is last", input: "aabbbb", expected: 4 },
  { name: "empty", input: "", expected: 0 },
];

test.each(inputs)("$name", ({ input, expected }) => {   // $name 取对象的 name 字段
  expect(maxBlock(input)).toBe(expected);
});

测试异常

第 1 章的第一个测试就是「期待异常」的测试。三种写法：

@Test
void rejectsNull() {
  assertThatThrownBy(() -> new BlockCounter().maxBlock(null))   // lambda：一段待执行的代码
      .isInstanceOf(NullPointerException.class);
}

def test_rejects_non_strings():
    with pytest.raises(TypeError):    # with 块里必须抛出指定异常，否则测试失败
        max_block(42)                 # 42[0] → TypeError

# 注意：max_block(None) 并不会抛异常！`if not s` 把 None 也当成了空字符串，返回 0。
# 这到底是 Bug 还是特性？这是一个契约问题（第 4 章）—— 测试帮你把它暴露了出来。

test("rejects null", () => {
  // 传一个函数给 expect，而不是直接调用；否则异常会在 expect 之前就抛出
  expect(() => maxBlock(null as unknown as string)).toThrow(TypeError);
});

小结

  • 一个测试 = 准备 + 执行 + 断言。JUnit 用 @Test 标记，pytest 用 test_ 前缀，Vitest 用 test()。
  • 失败报告告诉你期望值和实际值；学会读它，是调试的第一步。
  • 生命周期钩子（@BeforeEach / fixture / beforeEach）让每个测试都从干净、新鲜的状态开始。
  • 结构相同、数据不同的测试，用参数化测试写成一张表。
  • 准备好了？从第 1 章开始吧。
